核心源码：love.js

本项目唯一的依赖是 你。以下代码不需要 Review，因为答案显而易见。

// love.js —— 运行环境：你和我
const me   = createPerson("喜欢你的那个人");
const you  = createPerson("ok姐");   // 她，唯一的她

function love(me, you) {
  while (true) {
    me.heart.beatsOnlyFor(you);      // 心跳只为你
    me.day.increment();
    if (you.isHappy()) me.rejoice();
    else me.stay();                  // 陪着你，不打扰
    // 注释：没有退出条件
    // 也不需要退出条件
  }
}

love(me, you);  // 调用一次，运行一生

// 复杂度：O(一辈子) · 空间复杂度：两个人刚刚好
